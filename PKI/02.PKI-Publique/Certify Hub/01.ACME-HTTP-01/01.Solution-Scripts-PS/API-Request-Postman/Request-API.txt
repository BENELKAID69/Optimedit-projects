02.Script =

-------
const response = pm.response.json();
if (response.isSuccess) {
    pm.environment.set("access_token", response.accessToken);
    pm.environment.set("refresh_token", response.refreshToken);
    
    const expiresInSeconds = 3600;
    const expiryTimestamp = Date.now() + (expiresInSeconds * 1000);
    pm.environment.set("token_expiry", expiryTimestamp);
    
    console.log("Token stocké, expire à :", new Date(expiryTimestamp).toLocaleTimeString());
} else {
    console.error("Login échoué :", response.detail);
}
